<script>
  addEventListener("DOMContentLoaded", () => {

    const CONFIG = {
      hold: 3,
      move: 0.55,
      startDelay: 2000, // ⏱️ ms before first init

      lockWidthDesktop: true,
      lockWidthTablet: false,
      lockWidthMobile: false
    };

    let hasStarted = false; // 🚦 prevents early init

    function getBreakpoint() {
      const w = window.innerWidth;
      if (w <= 479) return "mobile";
      if (w <= 991) return "tablet";
      return "desktop";
    }

    function teardown(root) {
      if (root._verticalScrollTl) {
        root._verticalScrollTl.kill();
        root._verticalScrollTl = null;
      }
    }

    function removeClone(track) {
      if (track.dataset.cloned === "true" && track.lastElementChild?.classList.contains("rotate-text")) {
        track.removeChild(track.lastElementChild);
        track.dataset.cloned = "false";
      }
    }

    function shouldLockWidth() {
      const bp = getBreakpoint();
      if (bp === "desktop") return CONFIG.lockWidthDesktop;
      if (bp === "tablet") return CONFIG.lockWidthTablet;
      return CONFIG.lockWidthMobile;
    }

    function initWordCarousel(root) {
      const swap  = root.querySelector(".vertical-scroll-wrapper");
      const track = root.querySelector(".track-vertical");
      if (!swap || !track) return;

      teardown(root);
      removeClone(track);

      const words = Array.from(track.querySelectorAll(".rotate-text"));
      if (words.length < 2) return;

      gsap.set(track, { y: 0 });

      const cellHeight = words[0].getBoundingClientRect().height;
      swap.style.height = Math.ceil(cellHeight) + "px";

      if (shouldLockWidth()) {
        const maxWidth = Math.max(...words.map(w => w.getBoundingClientRect().width));
//        swap.style.width = Math.ceil(maxWidth) + "px";
      } else {
        swap.style.width = "";
      }

      track.appendChild(words[0].cloneNode(true));
      track.dataset.cloned = "true";

      const total = words.length;
      const tl = gsap.timeline({ repeat: -1 });

      for (let i = 1; i <= total; i++) {
        tl.to(track, {
          y: -i * cellHeight,
          duration: CONFIG.move,
          ease: "power2.inOut"
        }).to({}, { duration: CONFIG.hold });
      }

      tl.set(track, { y: 0 });
      root._verticalScrollTl = tl;
    }

    function initAll() {
      if (!hasStarted) return; // 🚫 block early calls
      document.querySelectorAll(".text-rotation-wrapper").forEach(initWordCarousel);
    }

    // ⏱️ Delayed start (ONLY place init is allowed initially)
    setTimeout(() => {
      hasStarted = true;
      initAll();
    }, CONFIG.startDelay);

    // Fonts ready — allowed only AFTER delay
    if (document.fonts?.ready) {
      document.fonts.ready.then(initAll);
    } else {
      addEventListener("load", initAll);
    }

    // Resize-safe reinit
    let t;
    addEventListener("resize", () => {
      if (!hasStarted) return;
      clearTimeout(t);
      t = setTimeout(initAll, 150);
    });

    // ResizeObserver
    if ("ResizeObserver" in window) {
      const ro = new ResizeObserver(() => {
        if (!hasStarted) return;
        clearTimeout(t);
        t = setTimeout(initAll, 50);
      });

      document.querySelectorAll(".text-rotation-wrapper").forEach(el => ro.observe(el));
    }

  });
</script>
